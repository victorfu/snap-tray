cmake_minimum_required(VERSION 3.20)
project(SnapTrayAppImageUpdater LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UPDATER_GPG REQUIRED IMPORTED_TARGET gpgme gpg-error)
include(FetchContent)
set(BUILD_LIBAPPIMAGEUPDATE_ONLY ON CACHE BOOL "" FORCE)
set(BUILD_QT_UI OFF CACHE BOOL "" FORCE)
set(LIBAPPIMAGE_SHARED_ONLY ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(appimageupdate
    GIT_REPOSITORY https://github.com/AppImageCommunity/AppImageUpdate.git
    GIT_TAG a211784dfc746fdb6d8d32d6bb39add451c1ddeb)
FetchContent_MakeAvailable(appimageupdate)
add_executable(snaptray-appimage-updater main.cpp)
target_link_libraries(snaptray-appimage-updater PRIVATE Qt6::Core Qt6::Network libappimageupdate_static signing util PkgConfig::UPDATER_GPG)
set_target_properties(snaptray-appimage-updater PROPERTIES BUILD_RPATH_USE_ORIGIN ON)

option(SNAPTRAY_BUILD_UPDATER_TESTS "Build isolated updater transaction test runner" OFF)
if(SNAPTRAY_BUILD_UPDATER_TESTS)
    add_executable(snaptray-appimage-updater-test main.cpp)
    target_compile_definitions(snaptray-appimage-updater-test PRIVATE SNAPTRAY_UPDATER_TEST_RUNNER)
    target_link_libraries(snaptray-appimage-updater-test PRIVATE Qt6::Core Qt6::Network libappimageupdate_static signing util PkgConfig::UPDATER_GPG)
endif()
